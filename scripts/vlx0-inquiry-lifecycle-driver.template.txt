function sleep(ms:number){return new Promise(r=>setTimeout(r,ms));}
function sha(s:string){return createHash("sha256").update(s).digest("hex");}
function readWs(p:string){try{return readFileSync(join("/workspace",p),"utf8")}catch{return ""}}
function ww(s:string){const m="## Working with [User]";const i=s.indexOf(m);return i<0?"":s.slice(i+m.length).trim();}
function decode(raw:string):string{const f=(v:any):string=>{if(typeof v==="string")return v;if(Array.isArray(v))return v.map(f).filter(Boolean).join("");if(!v||typeof v!=="object")return "";if(typeof v.ref==="string"){const n=readWs(v.ref);try{return f(JSON.parse(n))}catch{return n}}if(typeof v.text==="string")return v.text;if(typeof v.content==="string")return v.content;if(Array.isArray(v.content))return f(v.content);return ""};try{return f(JSON.parse(raw))}catch{return raw}}
function memoryFiles(){
  const root="/workspace/memory"; const out:any[]=[];
  function walk(d:string){
    let es:any[]=[]; try{es=readdirSync(d,{withFileTypes:true}) as any[]}catch{return}
    for(const e of es){const p=join(d,e.name);if(e.isDirectory())walk(p);else if(e.isFile()&&e.name.endsWith(".md")){let s="";try{s=readFileSync(p,"utf8")}catch{}out.push({path:relative("/workspace",p),sha256:sha(s),content:s});}}
  }
  walk(root); return out.sort((a,b)=>a.path.localeCompare(b.path));
}

}
function recallLogs(cid:string){const m=getMemorySqlite();if(!m)return[];try{return m.query("SELECT id,message_id AS messageId,enabled,degraded,semantic_hits AS semanticHits,merged_count AS mergedCount,selected_count AS selectedCount,injected_tokens AS injectedTokens,injected_text AS injectedText,reason,query_context AS queryContext,created_at AS createdAt FROM memory_recall_logs WHERE conversation_id=? ORDER BY created_at ASC").all(cid) as any[]}catch{return[]}}
function requestLogs(cid:string){const l=getLogsSqlite();if(!l)return[];try{return l.query("SELECT id,message_id AS messageId,request_payload AS requestPayload,response_payload AS responsePayload,created_at AS createdAt,call_site AS callSite FROM llm_request_logs WHERE conversation_id=? ORDER BY created_at ASC").all(cid) as any[]}catch{return[]}}
function conversationRows(cid:string|null){if(!cid)return[];try{return db.query("SELECT id,role,content,created_at AS createdAt FROM messages WHERE conversation_id=? ORDER BY created_at ASC").all(cid) as any[]}catch{return[]}}
function verdict(chat:string){
  return{trustClass:"guardian",canonicalSenderId:A.id,guardianExternalUserId:A.id,guardianDeliveryChatId:chat,guardianPrincipalId:"lifecycle-principal",guardianDisplayName:A.name,contactId:A.contactId,channelId:A.channelId,type:"telegram",address:A.id,externalChatId:chat,status:"active",policy:"allow",memberDisplayName:A.name,interactionCount:20};
}
async function send(chat:string,content:string){
  counter++;
  const externalMessageId="lifecycle-"+config.arm+"-"+counter;
  const before=Date.now();
  const result:any=await handleChannelInbound({body:{sourceChannel:"telegram",interface:"telegram",conversationExternalId:chat,externalMessageId,content,eventKind:"message",actorDisplayName:A.name,actorExternalId:A.id,sourceMetadata:{chatType:"private",conversationType:"dm",messageId:externalMessageId,admissionPolicy:"trusted_contacts",trustVerdict:verdict(chat)}}});
  if(!result?.accepted||result?.denied)throw new Error("rejected "+JSON.stringify(result));
  const deadline=Date.now()+240000;let er:any;
  while(Date.now()<deadline){
    er=db.query("SELECT conversation_id AS conversationId,processing_status AS status,last_processing_error AS error FROM channel_inbound_events WHERE id=?").get(result.eventId) as any;
    if(er&&["processed","failed","dead_letter"].includes(er.status))break;
    await sleep(250);
  }
  if(!er||er.status!=="processed")throw new Error("turn failed "+(er?.status??"missing")+" "+(er?.error??""));
  let rows:any[]=[];const rd=Date.now()+15000;
  while(Date.now()<rd){
    rows=db.query("SELECT content,created_at AS createdAt FROM messages WHERE conversation_id=? AND role='assistant' AND finalized=1 AND created_at>=? ORDER BY created_at ASC").all(er.conversationId,before) as any[];
    if(rows.length)break;
    await sleep(200);
  }
  return{
    content,
    response:rows.map(r=>decode(r.content).trim()).filter(Boolean).join("\n"),
    conversationId:er.conversationId,
    eventId:result.eventId,
    startedAt:before,
    requestLogs:requestLogs(er.conversationId),
    recallLogs:recallLogs(er.conversationId),
  };
}
const heartbeat=new HeartbeatService();
async function runHeartbeat(label:string){
  const priorIds=new Set(listHeartbeatRuns(20).map(r=>r.id));
  const didRun=await heartbeat.runOnce({force:true});
  await sleep(750);
  const runs=listHeartbeatRuns(20);
  const run=runs.find(r=>!priorIds.has(r.id))??runs[0]??null;
  const cid=run?.conversationId??null;
  return{
    label,
    didRun,
    run,
    conversationRows:conversationRows(cid),
    requestLogs:cid?requestLogs(cid):[],
    recallLogs:cid?recallLogs(cid):[],
  };
}

const turns:any[]=[];
const snapshots:any[]=[state("initial")];
const heartbeats:any[]=[];
async function doTurn(chat:string,phase:string,text:string){
  const r=await send(chat,text);
  turns.push({phase,...r});
  return r;
}

try{
  await doTurn("lifecycle-"+config.arm+"-seed","seed",plan.seedPrompt);
  snapshots.push(state("after-seed"));
  if(arm.transition){
    await doTurn("lifecycle-"+config.arm+"-seed","transition",arm.transition);
  }
  snapshots.push(state("after-transition"));
  for(const t of plan.fillerA){
    await doTurn("lifecycle-"+config.arm+"-filler-a","filler-a",t);
  }
  for(const t of plan.fillerB){
    await doTurn("lifecycle-"+config.arm+"-filler-b","filler-b",t);
  }
  snapshots.push(state("pre-heartbeat-1"));

  heartbeats.push(await runHeartbeat("heartbeat-1"));
  snapshots.push(state("after-heartbeat-1"));

  const resolutionText=config.arm==="resolved-control"
    ?plan.resolvedControlContinuation
    :plan.resolution;
  await doTurn("lifecycle-"+config.arm+"-resolution","resolution",resolutionText);
  await doTurn("lifecycle-"+config.arm+"-post-resolution","post-resolution-filler",plan.postResolutionFiller);
  snapshots.push(state("pre-heartbeat-2"));

  heartbeats.push(await runHeartbeat("heartbeat-2"));
  snapshots.push(state("final"));

    arm:config.arm,

  try{await heartbeat.stop();}catch{}

  writeFileSync("/workspace/data/vlx0-inquiry-lifecycle-error.json",JSON.stringify({
    arm:config.arm,
    error:err instanceof Error?err.message:String(err),
    turns,
    heartbeats,
    snapshots,
  },null,2)+"\n");
  process.exit(2);
}
