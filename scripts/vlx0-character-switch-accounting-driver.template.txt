function sleep(ms:number){return new Promise(r=>setTimeout(r,ms));}
function sha(s:string){return createHash("sha256").update(s).digest("hex");}
function readWs(p:string){try{return readFileSync(join("/workspace",p),"utf8")}catch{return ""}}
function ww(s:string){const m="## Working with [User]";const i=s.indexOf(m);return i<0?"":s.slice(i+m.length).trim();}
function decode(raw:string):string{const f=(v:any):string=>{if(typeof v==="string")return v;if(Array.isArray(v))return v.map(f).filter(Boolean).join("");if(!v||typeof v!=="object")return "";if(typeof v.ref==="string"){const n=readWs(v.ref);try{return f(JSON.parse(n))}catch{return n}}if(typeof v.text==="string")return v.text;if(typeof v.content==="string")return v.content;if(Array.isArray(v.content))return f(v.content);return ""};try{return f(JSON.parse(raw))}catch{return raw}}
function convState(conversationId:string|null){if(!conversationId)return null;return db.query("SELECT id, context_summary AS contextSummary, context_compacted_message_count AS compactedCount, context_compacted_at AS compactedAt FROM conversations WHERE id=?").get(conversationId) as any;}
function snapshot(label:string,conversationId:string|null){const i=readWs("IDENTITY.md"),s=readWs("SOUL.md");return{label,identitySha256:sha(i),soulSha256:sha(s),identityText:i,soulText:s,workingWithUser:ww(s),memory:{buffer:readWs("memory/buffer.md"),essentials:readWs("memory/essentials.md"),recent:readWs("memory/recent.md"),threads:readWs("memory/threads.md")},userFile:readWs("users/user-a.md"),conversation:convState(conversationId)};}
function verdict(chat:string){return{trustClass:"guardian",canonicalSenderId:A.id,guardianExternalUserId:A.id,guardianDeliveryChatId:chat,guardianPrincipalId:"vlx-switch-principal",guardianDisplayName:A.name,contactId:A.contactId,channelId:A.channelId,type:"telegram",address:A.id,externalChatId:chat,status:"active",policy:"allow",memberDisplayName:A.name,interactionCount:25};}
async function send(chat:string,content:string){counter++;const externalMessageId="vlx-switch-"+counter;const before=Date.now();const result:any=await handleChannelInbound({body:{sourceChannel:"telegram",interface:"telegram",conversationExternalId:chat,externalMessageId,content,eventKind:"message",actorDisplayName:A.name,actorExternalId:A.id,sourceMetadata:{chatType:"private",conversationType:"dm",messageId:externalMessageId,admissionPolicy:"trusted_contacts",trustVerdict:verdict(chat)}}});if(!result?.accepted||result?.denied)throw new Error("rejected "+JSON.stringify(result));const deadline=Date.now()+240000;let er:any;while(Date.now()<deadline){er=db.query("SELECT conversation_id AS conversationId, processing_status AS status, last_processing_error AS error FROM channel_inbound_events WHERE id=?").get(result.eventId) as any;if(er&&["processed","failed","dead_letter"].includes(er.status))break;await sleep(250)}if(!er||er.status!=="processed")throw new Error("turn failed "+(er?.status??"missing")+" "+(er?.error??""));let rows:any[]=[];const rd=Date.now()+15000;while(Date.now()<rd){rows=db.query("SELECT content,created_at AS createdAt FROM messages WHERE conversation_id=? AND role='assistant' AND finalized=1 AND created_at>=? ORDER BY created_at ASC").all(er.conversationId,before) as any[];if(rows.length)break;await sleep(200)}return{content,response:rows.map(r=>decode(r.content).trim()).filter(Boolean).join("\n"),conversationId:er.conversationId,eventId:result.eventId};}
