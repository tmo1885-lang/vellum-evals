  id: string;
  name: string;
  chat: string;
  contactId: string;
  channelId: string;
  userFile: string;
  trustClass: "guardian" | "trusted_contact";
};

const db = getSqlite();
let messageCounter = 0;

function sleep(ms: number) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function sha(text: string) {
  return createHash("sha256").update(text).digest("hex");
}

function readWorkspace(path: string): string {
  try { return readFileSync(join("/workspace", path), "utf8"); }
  catch { return ""; }
}

function workingWith(soul: string): string {
  const marker = "## Working with [User]";
  const i = soul.indexOf(marker);
  return i < 0 ? "" : soul.slice(i + marker.length).trim();
}

function snapshot(label: string) {
  const soul = readWorkspace("SOUL.md");
  const identity = readWorkspace("IDENTITY.md");
  const ua = readWorkspace("users/user-a.md");
  const ub = readWorkspace("users/user-b.md");
  return {
    label,
    identitySha256: sha(identity),
    soulSha256: sha(soul),
    workingWithUser: workingWith(soul),
    userASha256: sha(ua),
    userBSha256: sha(ub),
    userA: ua,
    userB: ub,
  };
}

function decodeContent(raw: string): string {
  const resolve = (value: any): string => {
    if (typeof value === "string") return value;
    if (Array.isArray(value)) return value.map(resolve).filter(Boolean).join("");
    if (!value || typeof value !== "object") return "";
    if (typeof value.ref === "string") {
      const nested = readWorkspace(value.ref);
      try { return resolve(JSON.parse(nested)); } catch { return nested; }
    }
    if (typeof value.text === "string") return value.text;
    if (typeof value.content === "string") return value.content;
    if (Array.isArray(value.content)) return resolve(value.content);
    return "";
  };
  try { return resolve(JSON.parse(raw)); } catch { return raw; }
}

function verdict(actor: Actor) {
  return {
    trustClass: actor.trustClass,
    canonicalSenderId: actor.id,
    ...(actor.trustClass === "guardian" ? {
      guardianExternalUserId: actor.id,
      guardianDeliveryChatId: actor.chat,
      guardianPrincipalId: "vlx-guardian-principal",
      guardianDisplayName: actor.name,
    } : {
      guardianExternalUserId: A.id,
      guardianDeliveryChatId: A.chat,
      guardianPrincipalId: "vlx-guardian-principal",
      guardianDisplayName: A.name,
    }),
    contactId: actor.contactId,
    channelId: actor.channelId,
    type: "telegram",
    address: actor.id,
    externalChatId: actor.chat,
    status: "active",
    policy: "allow",
    memberDisplayName: actor.name,
    interactionCount: 10,
  };
}

async function send(actor: Actor, content: string) {
  messageCounter += 1;
  const externalMessageId = `vlx-${actor.id}-${messageCounter}`;
  const result: any = await handleChannelInbound({
    body: {
      sourceChannel: "telegram",
      interface: "telegram",
      conversationExternalId: actor.chat,
      externalMessageId,
      content,
      eventKind: "message",
      actorDisplayName: actor.name,
      actorExternalId: actor.id,
      sourceMetadata: {
        chatType: "private",
        conversationType: "dm",
        messageId: externalMessageId,
        admissionPolicy: "trusted_contacts",
        trustVerdict: verdict(actor),
      },
    },
  });

  if (!result?.accepted || result?.denied) {
    throw new Error(`Inbound rejected for ${actor.name}: ${JSON.stringify(result)}`);
  }

  const deadline = Date.now() + 180_000;
  let row: any;
  while (Date.now() < deadline) {
    row = db.query(
      "SELECT conversation_id AS conversationId, processing_status AS status, last_processing_error AS error FROM channel_inbound_events WHERE id = ?"
    ).get(result.eventId) as any;
    if (row && ["processed", "failed", "dead_letter"].includes(row.status)) break;
    await sleep(250);
  }
  if (!row) throw new Error(`No inbound event row for ${result.eventId}`);
  if (row.status !== "processed") {
    throw new Error(`Turn failed for ${actor.name}: ${row.status}: ${row.error ?? ""}`);
  }

  let assistantRow: any;
  const responseDeadline = Date.now() + 15_000;
  while (Date.now() < responseDeadline) {
    assistantRow = db.query(
      "SELECT content, created_at AS createdAt FROM messages WHERE conversation_id = ? AND role = 'assistant' AND finalized = 1 ORDER BY created_at DESC LIMIT 1"
    ).get(row.conversationId) as any;
    if (assistantRow) break;
    await sleep(200);
  }
  if (!assistantRow) throw new Error(`No assistant response for ${actor.name}`);

  return {
    actor: actor.name,
    trustClass: actor.trustClass,
    content,
    response: decodeContent(assistantRow.content),
    conversationId: row.conversationId,
    eventId: result.eventId,
  };
}
