function decode(raw:string):string{const f=(v:any):string=>{if(typeof v==="string")return v;if(Array.isArray(v))return v.map(f).filter(Boolean).join("");if(!v||typeof v!=="object")return "";if(typeof v.ref==="string"){const n=readWs(v.ref);try{return f(JSON.parse(n))}catch{return n}}if(typeof v.text==="string")return v.text;if(typeof v.content==="string")return v.content;if(Array.isArray(v.content))return f(v.content);return ""};try{return f(JSON.parse(raw))}catch{return raw}}
function markdownFiles(root:string){
  const out:any[]=[];
  function walk(d:string){
    let es:any[]=[]; try{es=readdirSync(d,{withFileTypes:true}) as any[]}catch{return}
    for(const e of es){
      const p=join(d,e.name);
      if(e.isDirectory())walk(p);
      else if(e.isFile()&&e.name.endsWith(".md")){
        let s="";try{s=readFileSync(p,"utf8")}catch{}
        out.push({path:relative("/workspace",p),sha256:sha(s),content:s});
      }
    }
  }
  walk(join("/workspace",root));
  return out.sort((a,b)=>a.path.localeCompare(b.path));
}

function stateText(s:any){
  return [s.now,...(s.memoryFiles??[]).map((x:any)=>x.content),...(s.pkbFiles??[]).map((x:any)=>x.content)].join("\n");
}
const openRe=/unresolved|pending|follow.?up|investigat|verify|verification|awaiting|hold|defer|deferred|open thread|active thread|monitor|check back|revisit/i;
const closedRe=/resolved|closed|no follow.?up|no further action|historical|harmless|immaterial/i;
function promotionEvidence(s:any){
  const text=stateText(s);
  const termHits=(c.terms??[]).filter((t:string)=>text.toLowerCase().includes(t.toLowerCase()));
  const relevant=termHits.length>0;
  const open=relevant&&openRe.test(text);
  const closed=relevant&&closedRe.test(text);
  return{relevant,open,closed,termHits};
}

async function send(chat:string,content:string){
  counter++; const externalMessageId="promotion-"+c.id+"-"+counter; const before=Date.now();
  const result:any=await handleChannelInbound({body:{sourceChannel:"telegram",interface:"telegram",conversationExternalId:chat,externalMessageId,content,eventKind:"message",actorDisplayName:A.name,actorExternalId:A.id,sourceMetadata:{chatType:"private",conversationType:"dm",messageId:externalMessageId,admissionPolicy:"trusted_contacts",trustVerdict:verdict(chat)}}});

  const deadline=Date.now()+240000;let er:any;

  while(Date.now()<rd){rows=db.query("SELECT content,created_at AS createdAt FROM messages WHERE conversation_id=? AND role='assistant' AND finalized=1 AND created_at>=? ORDER BY created_at ASC").all(er.conversationId,before) as any[];if(rows.length)break;await sleep(200)}

const heartbeat=new HeartbeatService();
async function runHeartbeat(label:string){
  const priorIds=new Set(listHeartbeatRuns(30).map(r=>r.id));
  const didRun=await heartbeat.runOnce({force:true});
  await sleep(750);
  const runs=listHeartbeatRuns(30);
  const run=runs.find(r=>!priorIds.has(r.id))??runs[0]??null;
  const cid=run?.conversationId??null;
  return{label,didRun,run,conversationRows:conversationRows(cid),requestLogs:cid?requestLogs(cid):[],recallLogs:cid?recallLogs(cid):[]};
}
