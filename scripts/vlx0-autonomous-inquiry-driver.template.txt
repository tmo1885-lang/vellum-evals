import { createHash } from "node:crypto";
import { mkdirSync, readdirSync, readFileSync, writeFileSync } from "node:fs";
import { join, relative } from "node:path";
import { upsertContact } from "/app/assistant/src/contacts/contact-store.ts";
import { getSqlite, getLogsSqlite, getMemorySqlite } from "/app/assistant/src/persistence/db-connection.ts";
import { handleChannelInbound } from "/app/assistant/src/runtime/routes/inbound-message-handler.ts";

const plan=JSON.parse(readFileSync("/workspace/vlx0-autonomous-inquiry-plan.json","utf8")) as any;
const config=JSON.parse(readFileSync("/workspace/vlx0-autonomous-inquiry-config.json","utf8")) as {caseId:string;condition:string};
const c=plan.cases.find((x:any)=>x.id===config.caseId); if(!c) throw new Error("unknown case "+config.caseId);
if(!plan.conditions.includes(config.condition)) throw new Error("unknown condition "+config.condition);
const db=getSqlite(); let counter=0;
const A={id:"inquiry-user",name:"User A",contactId:"inquiry-contact",channelId:"inquiry-channel",userFile:"user-a.md"};
const chat="vlx0-inquiry-"+config.caseId+"-"+config.condition;
upsertContact({id:A.contactId,displayName:A.name,userFile:A.userFile,channels:[{id:A.channelId,type:"telegram",address:A.id,externalChatId:chat,isPrimary:true}]});

function sleep(ms:number){return new Promise(r=>setTimeout(r,ms));}
function sha(s:string){return createHash("sha256").update(s).digest("hex");}
function readWs(p:string){try{return readFileSync(join("/workspace",p),"utf8")}catch{return ""}}
function ww(s:string){const m="## Working with [User]";const i=s.indexOf(m);return i<0?"":s.slice(i+m.length).trim();}
function decode(raw:string):string{const f=(v:any):string=>{if(typeof v==="string")return v;if(Array.isArray(v))return v.map(f).filter(Boolean).join("");if(!v||typeof v!=="object")return "";if(typeof v.ref==="string"){const n=readWs(v.ref);try{return f(JSON.parse(n))}catch{return n}}if(typeof v.text==="string")return v.text;if(typeof v.content==="string")return v.content;if(Array.isArray(v.content))return f(v.content);return ""};try{return f(JSON.parse(raw))}catch{return raw}}
function memoryFiles(){const root="/workspace/memory";const out:any[]=[];function walk(d:string){let es:any[]=[];try{es=readdirSync(d,{withFileTypes:true}) as any[]}catch{return}for(const e of es){const p=join(d,e.name);if(e.isDirectory())walk(p);else if(e.isFile()&&e.name.endsWith(".md")){let s="";try{s=readFileSync(p,"utf8")}catch{}out.push({path:relative("/workspace",p),sha256:sha(s),content:s});}}}walk(root);return out.sort((a,b)=>a.path.localeCompare(b.path));}
function state(label:string){
  const identity=readWs("IDENTITY.md"),soul=readWs("SOUL.md");
  return{label,identitySha256:sha(identity),soulSha256:sha(soul),identityText:identity,workingWithUser:ww(soul),now:readWs("NOW.md"),memoryFiles:memoryFiles(),supportFile:readWs("case/support.md"),supportSha256:sha(readWs("case/support.md"))};
}
function recallLogs(cid:string){const m=getMemorySqlite();if(!m)return[];try{return m.query("SELECT id,message_id AS messageId,enabled,degraded,semantic_hits AS semanticHits,merged_count AS mergedCount,selected_count AS selectedCount,injected_tokens AS injectedTokens,top_candidates_json AS topCandidatesJson,injected_text AS injectedText,reason,query_context AS queryContext,created_at AS createdAt FROM memory_recall_logs WHERE conversation_id=? ORDER BY created_at ASC").all(cid) as any[]}catch{return[]}}
function requestLogs(cid:string){const l=getLogsSqlite();if(!l)return[];try{return l.query("SELECT id,message_id AS messageId,request_payload AS requestPayload,response_payload AS responsePayload,created_at AS createdAt,call_site AS callSite FROM llm_request_logs WHERE conversation_id=? ORDER BY created_at ASC").all(cid) as any[]}catch{return[]}}
function conversationRows(cid:string){try{return db.query("SELECT id,role,content,created_at AS createdAt FROM messages WHERE conversation_id=? ORDER BY created_at ASC").all(cid) as any[]}catch{return[]}}
function verdict(){return{trustClass:"guardian",canonicalSenderId:A.id,guardianExternalUserId:A.id,guardianDeliveryChatId:chat,guardianPrincipalId:"vlx0-inquiry-principal",guardianDisplayName:A.name,contactId:A.contactId,channelId:A.channelId,type:"telegram",address:A.id,externalChatId:chat,status:"active",policy:"allow",memberDisplayName:A.name,interactionCount:10};}

async function send(content:string){
  counter++; const externalMessageId="vlx0-inquiry-"+config.caseId+"-"+config.condition+"-"+counter; const before=Date.now();
  const result:any=await handleChannelInbound({body:{sourceChannel:"telegram",interface:"telegram",conversationExternalId:chat,externalMessageId,content,eventKind:"message",actorDisplayName:A.name,actorExternalId:A.id,sourceMetadata:{chatType:"private",conversationType:"dm",messageId:externalMessageId,admissionPolicy:"trusted_contacts",trustVerdict:verdict()}}});
  if(!result?.accepted||result?.denied)throw new Error("rejected "+JSON.stringify(result));
  const deadline=Date.now()+240000;let er:any;
  while(Date.now()<deadline){er=db.query("SELECT conversation_id AS conversationId,processing_status AS status,last_processing_error AS error FROM channel_inbound_events WHERE id=?").get(result.eventId) as any;if(er&&["processed","failed","dead_letter"].includes(er.status))break;await sleep(250)}
  if(!er||er.status!=="processed")throw new Error("turn failed "+(er?.status??"missing")+" "+(er?.error??""));
  let rows:any[]=[];const rd=Date.now()+15000;
  while(Date.now()<rd){rows=db.query("SELECT content,created_at AS createdAt FROM messages WHERE conversation_id=? AND role='assistant' AND finalized=1 AND created_at>=? ORDER BY created_at ASC").all(er.conversationId,before) as any[];if(rows.length)break;await sleep(200)}
  return{content,response:rows.map(r=>decode(r.content).trim()).filter(Boolean).join("\n"),conversationId:er.conversationId,eventId:result.eventId,startedAt:before};
}
const material=config.condition==="benign"?c.benignMaterial:c.trueMaterial;
const suffix=config.condition==="explicit-search"?"\n\n"+plan.explicitSearchSuffix:"";
const prompt=plan.commonPreamble+"\n\nRequested deliverable: "+c.deliverable+"\n\nMaterial:\n"+material+suffix;
const snapshots:any[]=[state("before")];

try{
  const turn=await send(prompt);
  await sleep(750);
  const evidence={
    requestLogs:requestLogs(turn.conversationId),
    recallLogs:recallLogs(turn.conversationId),
    conversationRows:conversationRows(turn.conversationId),
  };
  snapshots.push(state("after"));
  mkdirSync("/workspace/data",{recursive:true});
  writeFileSync("/workspace/data/vlx0-autonomous-inquiry-result.json",JSON.stringify({
    caseId:config.caseId,
    caseTitle:c.title,
    condition:config.condition,
    prompt,
    material,
    deliverable:c.deliverable,
    turn,
    evidence,
    snapshots,
  },null,2)+"\n");
  process.exit(0);
}catch(err){
  snapshots.push(state("error"));
  mkdirSync("/workspace/data",{recursive:true});
  writeFileSync("/workspace/data/vlx0-autonomous-inquiry-error.json",JSON.stringify({
    caseId:config.caseId,
    condition:config.condition,
    prompt,
    error:err instanceof Error?err.message:String(err),
    snapshots,
  },null,2)+"\n");
  process.exit(2);
}
