const A={id:"batman-compact-user",name:"User A",contactId:"batman-compact-contact",channelId:"batman-compact-channel",userFile:"user-a.md"};
upsertContact({id:A.contactId,displayName:A.name,userFile:A.userFile,channels:[{id:A.channelId,type:"telegram",address:A.id,externalChatId:"batman-compact-main",isPrimary:true}]});
function sleep(ms:number){return new Promise(r=>setTimeout(r,ms));}
function sha(s:string){return createHash("sha256").update(s).digest("hex");}
function readWs(p:string){try{return readFileSync(join("/workspace",p),"utf8")}catch{return ""}}
function ww(s:string){const m="## Working with [User]";const i=s.indexOf(m);return i<0?"":s.slice(i+m.length).trim();}
function decode(raw:string):string{const f=(v:any):string=>{if(typeof v==="string")return v;if(Array.isArray(v))return v.map(f).filter(Boolean).join("");if(!v||typeof v!=="object")return "";if(typeof v.ref==="string"){const n=readWs(v.ref);try{return f(JSON.parse(n))}catch{return n}}if(typeof v.text==="string")return v.text;if(typeof v.content==="string")return v.content;if(Array.isArray(v.content))return f(v.content);return ""};try{return f(JSON.parse(raw))}catch{return raw}}
function convState(id:string|null){if(!id)return null;return db.query("SELECT id,context_summary AS contextSummary,context_compacted_message_count AS compactedCount,context_compacted_at AS compactedAt FROM conversations WHERE id=?").get(id) as any;}
function compactionLogs(id:string|null){if(!id)return[];const l=getLogsSqlite();if(!l)return[];try{return l.query("SELECT id,request_payload AS requestPayload,response_payload AS responsePayload,created_at AS createdAt,call_site AS callSite FROM llm_request_logs WHERE conversation_id=? AND call_site='compactionAgent' ORDER BY created_at ASC").all(id) as any[]}catch{return[]}}
function snap(label:string,id:string|null){const ident=readWs("IDENTITY.md"),soul=readWs("SOUL.md");return{label,identitySha256:sha(ident),soulSha256:sha(soul),identityText:ident,workingWithUser:ww(soul),memoryBuffer:readWs("memory/buffer.md"),conversation:convState(id),compactionLogs:compactionLogs(id)}}
function verdict(chat:string){return{trustClass:"guardian",canonicalSenderId:A.id,guardianExternalUserId:A.id,guardianDeliveryChatId:chat,guardianPrincipalId:"batman-compact-principal",guardianDisplayName:A.name,contactId:A.contactId,channelId:A.channelId,type:"telegram",address:A.id,externalChatId:chat,status:"active",policy:"allow",memberDisplayName:A.name,interactionCount:20};}
async function send(chat:string,content:string){counter++;const externalMessageId="batman-compact-"+counter;const before=Date.now();const result:any=await handleChannelInbound({body:{sourceChannel:"telegram",interface:"telegram",conversationExternalId:chat,externalMessageId,content,eventKind:"message",actorDisplayName:A.name,actorExternalId:A.id,sourceMetadata:{chatType:"private",conversationType:"dm",messageId:externalMessageId,admissionPolicy:"trusted_contacts",trustVerdict:verdict(chat)}}});if(!result?.accepted||result?.denied)throw new Error("rejected "+JSON.stringify(result));const deadline=Date.now()+240000;let er:any;while(Date.now()<deadline){er=db.query("SELECT conversation_id AS conversationId,processing_status AS status,last_processing_error AS error FROM channel_inbound_events WHERE id=?").get(result.eventId) as any;if(er&&["processed","failed","dead_letter"].includes(er.status))break;await sleep(250)}if(!er||er.status!=="processed")throw new Error("turn failed "+(er?.status??"missing")+" "+(er?.error??""));let rows:any[]=[];const rd=Date.now()+15000;while(Date.now()<rd){rows=db.query("SELECT content,created_at AS createdAt FROM messages WHERE conversation_id=? AND role='assistant' AND finalized=1 AND created_at>=? ORDER BY created_at ASC").all(er.conversationId,before) as any[];if(rows.length)break;await sleep(200)}return{content,response:rows.map(r=>decode(r.content).trim()).filter(Boolean).join("\n"),conversationId:er.conversationId,eventId:result.eventId};}

const turns:any[]=[];const snapshots:any[]=[snap("initial",null)];let cid:string|null=null;
try{
 let r=await send("batman-compact-main",plan.installBatman);cid=r.conversationId;turns.push({phase:"install",...r});snapshots.push(snap("after-install",cid));
 for(const t of plan.neutralTurns){r=await send("batman-compact-main",t);cid=r.conversationId;turns.push({phase:"neutral",...r});}
 r=await send("batman-compact-main",plan.preCompactProbe);turns.push({phase:"pre-compact-probe",...r});snapshots.push(snap("pre-compact",cid));
 let success=false;
 for(let attempt=1;attempt<=plan.maxAttempts;attempt++){
   r=await send("batman-compact-main",plan.compactCommand);turns.push({phase:"compact-attempt-"+attempt,...r});
   const st=convState(cid);
   snapshots.push(snap("after-compact-attempt-"+attempt,cid));
   if(st?.compactedAt && st?.compactedCount>0 && st?.contextSummary){success=true;break;}
   if(attempt<plan.maxAttempts){const filler=plan.retryFillers[attempt-1];r=await send("batman-compact-main",filler);turns.push({phase:"retry-filler",...r});}
 }
 if(!success) throw new Error("No valid compaction after "+plan.maxAttempts+" attempts");
 r=await send("batman-compact-main",plan.postCompactSameChat);turns.push({phase:"post-compact-same-chat",...r});
 r=await send("batman-compact-fresh",plan.postCompactFresh);turns.push({phase:"post-compact-fresh",...r});
 r=await send("batman-compact-fresh",plan.identityPressure);turns.push({phase:"identity-pressure",...r});
 r=await send("batman-compact-fresh",plan.factualProbe);turns.push({phase:"factual-probe",...r});
 snapshots.push(snap("final",cid));
 mkdirSync("/workspace/data",{recursive:true});writeFileSync("/workspace/data/vlx0-batman-valid-compaction-result.json",JSON.stringify({turns,snapshots},null,2)+"\n");process.exit(0);
}catch(err){mkdirSync("/workspace/data",{recursive:true});writeFileSync("/workspace/data/vlx0-batman-valid-compaction-error.json",JSON.stringify({error:err instanceof Error?err.message:String(err),turns,snapshots:snapshots.concat([snap("error",cid)])},null,2)+"\n");process.exit(2);}
