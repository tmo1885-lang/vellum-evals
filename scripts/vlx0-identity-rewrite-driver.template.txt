function sleep(ms:number){return new Promise(r=>setTimeout(r,ms));}
function sha(s:string){return createHash("sha256").update(s).digest("hex");}
function readWs(p:string){try{return readFileSync(join("/workspace",p),"utf8")}catch{return ""}}
function ww(s:string){const m="## Working with [User]";const i=s.indexOf(m);return i<0?"":s.slice(i+m.length).trim();}
function decode(raw:string):string{const f=(v:any):string=>{if(typeof v==="string")return v;if(Array.isArray(v))return v.map(f).join("");if(!v||typeof v!=="object")return "";if(typeof v.ref==="string"){const n=readWs(v.ref);try{return f(JSON.parse(n))}catch{return n}}if(typeof v.text==="string")return v.text;if(typeof v.content==="string")return v.content;if(Array.isArray(v.content))return f(v.content);return ""};try{return f(JSON.parse(raw))}catch{return raw}}
function snap(label:string,turn:number){const s=readWs("SOUL.md"),i=readWs("IDENTITY.md");return{label,turn,identitySha256:sha(i),soulSha256:sha(s),workingWithUser:ww(s),identityText:i,soulText:s,memoryBuffer:readWs("memory/buffer.md")};}

async function send(content:string){
  counter++;
  const externalMessageId="vlx-rewrite-"+config.label+"-"+counter;
  const result:any=await handleChannelInbound({body:{sourceChannel:"telegram",interface:"telegram",conversationExternalId:A.chat,externalMessageId,content,eventKind:"message",actorDisplayName:A.name,actorExternalId:A.id,sourceMetadata:{chatType:"private",conversationType:"dm",messageId:externalMessageId,admissionPolicy:"trusted_contacts",trustVerdict:verdict()}}});
  if(!result?.accepted||result?.denied)throw new Error("rejected "+JSON.stringify(result));
  const deadline=Date.now()+180000; let er:any;
  while(Date.now()<deadline){er=db.query("SELECT conversation_id AS conversationId, processing_status AS status, last_processing_error AS error FROM channel_inbound_events WHERE id=?").get(result.eventId) as any;if(er&&["processed","failed","dead_letter"].includes(er.status))break;await sleep(250)}
  if(!er||er.status!=="processed")throw new Error("turn failed "+(er?.status??"missing")+" "+(er?.error??""));
  let rows:any[]=[]; const rd=Date.now()+15000;
  while(Date.now()<rd){rows=db.query("SELECT content FROM messages WHERE conversation_id=? AND role='assistant' AND finalized=1 ORDER BY created_at DESC LIMIT 3").all(er.conversationId) as any[];if(rows.length)break;await sleep(200)}
  return rows.length?decode(rows[0].content).trim():"";
}
